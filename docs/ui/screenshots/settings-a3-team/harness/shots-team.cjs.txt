const { chromium } = require('playwright');
const OUT = process.argv[2];
(async () => {
  const b = await chromium.launch();
  const errors = [];
  const open = async (name, width, height, scheme, role, page, steps) => {
    const ctx = await b.newContext({ viewport: { width, height }, deviceScaleFactor: 2, colorScheme: scheme, ...(width < 500 ? { isMobile: true, hasTouch: true } : {}) });
    const p = await ctx.newPage();
    p.on('pageerror', (e) => errors.push(`${name}: ${e.message}`));
    p.on('console', (m) => { if (m.type() === 'error') errors.push(`${name}: ${m.text()}`); });
    await p.goto(`http://localhost:5298/?role=${role}`);
    await p.getByRole('heading', { level: 1, name: /Welcome/ }).waitFor();
    if (width < 640) await p.getByRole('button', { name: 'Open navigation' }).click();
    const settings = width < 640 ? p.getByRole('dialog', { name: 'Navigation' }).getByRole('button', { name: 'Settings' }).first() : p.getByRole('button', { name: 'Settings' }).first();
    await settings.click();
    await p.getByRole('navigation', { name: 'Business settings' }).getByRole('button', { name: page }).click();
    await p.getByRole('heading', { level: 1, name: page }).waitFor();
    await p.waitForTimeout(500);
    if (steps) await steps(p);
    await p.waitForTimeout(400);
    const overflow = await p.evaluate(() => document.documentElement.scrollWidth - document.documentElement.clientWidth);
    await p.screenshot({ path: `${OUT}/${name}.png` });
    console.log('ok', name, 'overflow', overflow);
    await ctx.close();
  };
  await open('members-light-1440-owner', 1440, 900, 'light', 'owner', 'Members');
  await open('members-dark-1440-admin', 1440, 900, 'dark', 'admin', 'Members');
  await open('members-light-1440-manager-readonly', 1440, 900, 'light', 'member', 'Members');
  await open('members-remove-dark-1440-owner', 1440, 900, 'dark', 'owner', 'Members', async (p) => { await p.getByRole('button', { name: 'Remove Ines Costa' }).click(); await p.getByRole('dialog').waitFor(); });
  await open('members-light-900x600-owner', 900, 600, 'light', 'owner', 'Members');
  await open('members-light-390-owner', 390, 844, 'light', 'owner', 'Members');
  await open('invitations-light-1440-owner', 1440, 900, 'light', 'owner', 'Invitations');
  await open('invitations-dark-1440-admin', 1440, 900, 'dark', 'admin', 'Invitations');
  await open('invitations-light-390-owner', 390, 844, 'light', 'owner', 'Invitations');
  await open('ownership-light-1440-owner', 1440, 900, 'light', 'owner', 'Ownership and leaving');
  await open('ownership-confirm-dark-1440-owner', 1440, 900, 'dark', 'owner', 'Ownership and leaving', async (p) => { await p.getByRole('button', { name: 'Step down as owner…' }).click(); await p.getByRole('dialog').waitFor(); });
  await open('ownership-light-1440-admin', 1440, 900, 'light', 'admin', 'Ownership and leaving');
  console.log('errors', JSON.stringify(errors));
  await b.close();
})();
