const { chromium } = require('playwright');
const OUT = process.argv[2];
(async () => {
  const b = await chromium.launch(); const errors = [];
  for (const [name, w, h, scheme, role] of [['zones-light-1440-owner',1440,900,'light','owner'],['zones-dark-1440-manager-readonly',1440,900,'dark','member'],['zones-light-390-owner',390,844,'light','owner']]) {
    const ctx = await b.newContext({ viewport: { width: w, height: h }, deviceScaleFactor: 2, colorScheme: scheme, ...(w < 500 ? { isMobile: true, hasTouch: true } : {}) });
    const p = await ctx.newPage(); p.on('pageerror', (e) => errors.push(e.message));
    await p.goto(`http://localhost:5298/?role=${role}`);
    await p.getByRole('heading', { level: 1, name: /Welcome/ }).waitFor();
    if (w < 640) await p.getByRole('button', { name: 'Open navigation' }).click();
    await (w < 640 ? p.getByRole('dialog', { name: 'Navigation' }).getByRole('button', { name: 'Settings' }).first() : p.getByRole('button', { name: 'Settings' }).first()).click();
    await p.getByRole('navigation', { name: 'Business settings' }).getByRole('button', { name: 'Zone configuration' }).click();
    await p.getByRole('heading', { level: 1, name: 'Zone configuration' }).waitFor(); await p.waitForTimeout(500);
    console.log(name, 'overflow', await p.evaluate(() => document.documentElement.scrollWidth - document.documentElement.clientWidth));
    await p.screenshot({ path: `${OUT}/${name}.png` }); await ctx.close();
  }
  console.log('errors', JSON.stringify(errors)); await b.close();
})();
