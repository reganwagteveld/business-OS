async function call(method, path, body, extra = {}) {
  const res = await fetch(`${API}${path}`, {
    method, headers: { origin: WEB, 'content-type': 'application/json', 'x-forwarded-for': `10.9.0.${ipn++ % 250}`, ...(cookie ? { cookie } : {}), ...extra },
    body: body === undefined ? undefined : JSON.stringify(body),
  });
  const set = res.headers.getSetCookie?.() ?? [];
  const sess = set.find((c) => /session_token=/.test(c));
  if (sess) cookie = sess.split(';')[0];
  const json = await res.json().catch(() => ({}));
  if (res.status >= 400) throw new Error(`${method} ${path} ${res.status} ${JSON.stringify(json)}`);
  return json;
}

const browser = await chromium.launch({ channel: 'chrome' });
process.on('unhandledRejection', () => {});
async function session(viewport, scheme) {
  const ctx = await browser.newContext({ viewport, deviceScaleFactor: 2, colorScheme: scheme, ...(viewport.width < 500 ? { isMobile: true, hasTouch: true } : {}) });
  const [name, value] = cookie.split('=');
  await ctx.addCookies([{ name, value, domain: 'localhost', path: '/', httpOnly: true, sameSite: 'Lax' }]);
  await ctx.addInitScript(([s]) => localStorage.setItem('bos.workspace', JSON.stringify({ slug: s })), [slug]);
  await ctx.route('http://localhost:5183/api/**', async (route) => {
    const r = route.request();
    const res = await route.fetch({ url: r.url().replace('http://localhost:5183', API), headers: { ...r.headers(), origin: WEB, cookie } });
    await route.fulfill({ response: res });
  });
  const page = await ctx.newPage();
  page.setDefaultTimeout(12000);
  await page.goto(WEB);
  await page.getByRole('button', { name: /Open the Finance Zone/ }).first().click();
  await page.getByText(/Outstanding on invoices \(AUD\)/).waitFor();
  await page.waitForTimeout(600);
  return { ctx, page };
}
