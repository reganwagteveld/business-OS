let ipn = 1;
function client() {
  const me = { cookie: '' };
  me.call = async (method, path, body, extra = {}) => {
    const res = await fetch(`${API}${path}`, {
      method, headers: { origin: WEB, 'content-type': 'application/json', 'x-forwarded-for': `10.9.${Math.floor(ipn / 250)}.${ipn++ % 250}`, ...(me.cookie ? { cookie: me.cookie } : {}), ...extra },
      body: body === undefined ? undefined : JSON.stringify(body),
    });
    const sess = (res.headers.getSetCookie?.() ?? []).find((c) => /session_token=/.test(c));
    if (sess) me.cookie = sess.split(';')[0];
    const json = await res.json().catch(() => ({}));
    if (res.status >= 400) throw new Error(`${method} ${path} ${res.status} ${JSON.stringify(json)}`);
    return json;
  };
  return me;
}

}

const browser = await chromium.launch({ channel: 'chrome' });
async function session(who, viewport, scheme) {
  const ctx = await browser.newContext({ viewport, deviceScaleFactor: 2, colorScheme: scheme, ...(viewport.width < 500 ? { isMobile: true, hasTouch: true } : {}) });
  const [name, value] = who.cookie.split('=');
  await ctx.addCookies([{ name, value, domain: 'localhost', path: '/', httpOnly: true, sameSite: 'Lax' }]);
  await ctx.addInitScript(([s]) => localStorage.setItem('bos.workspace', JSON.stringify({ slug: s })), [slug]);
  await ctx.route('http://localhost:5183/api/**', async (route) => {
    const r = route.request();
    const res = await route.fetch({ url: r.url().replace('http://localhost:5183', API), headers: { ...r.headers(), origin: WEB, cookie: who.cookie } });
    await route.fulfill({ response: res });
  });
  const page = await ctx.newPage();
  page.setDefaultTimeout(15000);
  const errors = [];
  page.on('console', (m) => { if (m.type() === 'error') errors.push(m.text()); });
  page.on('response', (r) => { if (r.status() >= 400) errors.push(`${r.request().method()} ${r.url()} ${r.status()}`); });
  await page.goto(WEB);
  await page.getByRole('button', { name: /Open the Reporting Zone/ }).first().click();
  await page.getByRole('region', { name: 'Headline figures' }).waitFor();
  await page.getByRole('heading', { name: 'Records' }).waitFor().catch(() => {});
  await page.waitForTimeout(900);
  return { ctx, page, errors };
}
// The shell scrolls its main column, not the document: grow the viewport to the content for full shots.
const shot = async (page, name, full = true) => {
  await page.evaluate(() => { window.scrollTo(0, 0); for (const el of document.querySelectorAll('*')) if (el.scrollTop) el.scrollTop = 0; });
  const size = page.viewportSize();
  if (full) {
    const h = await page.evaluate(() => Math.max(...[...document.querySelectorAll('*')].map((el) => el.scrollHeight)));
    await page.setViewportSize({ width: size.width, height: Math.min(h + 40, 6000) });
  }
  await page.waitForTimeout(400);
  await page.screenshot({ path: `${OUT}/${name}.png`, fullPage: full });
  await page.setViewportSize(size);
};
