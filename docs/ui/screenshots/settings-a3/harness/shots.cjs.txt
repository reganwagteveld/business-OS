const { chromium } = require('playwright');
const OUT = process.argv[2];
(async () => {
  const b = await chromium.launch();
  const errors = [];
  const open = async (name, width, height, scheme, role, steps) => {
    const ctx = await b.newContext({ viewport: { width, height }, deviceScaleFactor: 2, colorScheme: scheme, ...(width < 500 ? { isMobile: true, hasTouch: true } : {}) });
    const p = await ctx.newPage();
    p.on('pageerror', (e) => errors.push(`${name}: ${e.message}`));
    p.on('console', (m) => { if (m.type() === 'error') errors.push(`${name}: ${m.text()}`); });
    await p.goto(`http://localhost:5298/?role=${role}`);
    await p.getByRole('heading', { level: 1, name: /Welcome/ }).waitFor();
    if (width < 640) await p.getByRole('button', { name: 'Open navigation' }).click();
    const settings = width < 640 ? p.getByRole('dialog', { name: 'Navigation' }).getByRole('button', { name: 'Settings' }).first() : p.getByRole('button', { name: 'Settings' }).first();
    await settings.click();
    await steps(p);
    await p.waitForTimeout(500);
    const overflow = await p.evaluate(() => document.documentElement.scrollWidth - document.documentElement.clientWidth);
    await p.screenshot({ path: `${OUT}/${name}.png` });
    console.log('ok', name, 'overflow', overflow);
    await ctx.close();
  };
  const profile = async (p) => { await p.getByRole('heading', { level: 1, name: 'Business profile' }).waitFor(); await p.getByLabel('Business name').or(p.getByText('Details')).first().waitFor(); };
  const regional = async (p) => { await profile(p); await p.getByRole('button', { name: 'Regional settings' }).click(); await p.getByRole('heading', { level: 1, name: 'Regional settings' }).waitFor(); await p.waitForTimeout(400); };
  const account = async (p) => { await profile(p); await p.getByRole('button', { name: 'Account', exact: true }).first().click(); await p.getByRole('heading', { level: 1, name: 'Preferences' }).waitFor(); };
  await open('profile-light-1440-owner', 1440, 900, 'light', 'owner', profile);
  await open('profile-dark-1440-owner', 1440, 900, 'dark', 'owner', profile);
  await open('profile-light-1440-member-readonly', 1440, 900, 'light', 'member', profile);
  await open('regional-light-1440-owner', 1440, 900, 'light', 'owner', regional);
  await open('regional-confirm-dark-1440-owner', 1440, 900, 'dark', 'owner', async (p) => { await regional(p); await p.getByLabel('Default currency').selectOption('NZD'); await p.getByRole('button', { name: 'Save changes' }).click(); await p.getByRole('dialog').waitFor(); });
  await open('regional-light-900x600-owner', 900, 600, 'light', 'owner', regional);
  await open('regional-light-1440-member-readonly', 1440, 900, 'light', 'member', regional);
  await open('account-dark-1440', 1440, 900, 'dark', 'owner', account);
  await open('account-light-900x600', 900, 600, 'light', 'owner', account);
  await open('profile-light-390-owner', 390, 844, 'light', 'owner', profile);
  await open('profile-dark-390-member', 390, 844, 'dark', 'member', profile);
  console.log('errors', JSON.stringify(errors));
  await b.close();
})();
